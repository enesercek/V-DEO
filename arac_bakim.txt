create database arac_bakim;

use arac_bakim;

create table ustalar(
usta_no int identity (1,1)primary key,
usta_ad varchar(20),
usta_soyad varchar(20),
gorev varchar(30),
adres varchar(40),
tel int);

create table musteri(
musteri_no int identity(200,1) primary key,
musteri_adi varchar(20),
musteri_soyadi varchar(20),
adres varchar(40),
tel int);

create table arac(
arac_no int identity (1,1)primary key,
plaka_kodu varchar(10),
arac_yili datetime,
son_bakim_tarihi datetime,
musteri_no int,
constraint FK_arac_musteri
foreign key (musteri_no)
references musteri(musteri_no));

create table bakim(
bakim_no int identity(1,1) primary key, 
bakim_ucreti money,
bakim_baslangıc_tarih datetime,
bakim_bitis_tarih datetime,
arac_no int,
usta_no int,
constraint fk_bakim_arac
foreign key (arac_no)
references arac(arac_no),
constraint fk_bakim_usta
foreign key (usta_no)
references ustalar(usta_no));

create table yedek_parca(
parca_no int identity(1,1)primary key,
parca_adi varchar(30),
parca_maaliyeti money,
parca_satis_fiyati money,
bakim_no int,
constraint fk_parca_bakim
foreign key (bakim_no)
references bakim(bakim_no)
);

create table para(
para_id int identity(1,1) primary key,
bakim_no int,
parca_no int,
musteri_no int,
verilen_para money,
alinan_para money,

constraint fk_para_bakim
foreign key (bakim_no)
references bakim(bakim_no),
constraint fk_yedek_parca_para
foreign key (parca_no)
references yedek_parca(parca_no),
constraint fk_para_musteri
foreign key (musteri_no)
references musteri(musteri_no));



go

create procedure arac_tamir_suresi
(
@tarih1 datetime,
@tarih2 datetime
)
as 
begin
select
 datediff (day ,bakim_baslangıc_tarih,bakim_bitis_tarih)
from bakim

end



go
create procedure kar_miktarı
(
@para1 money,
@para2 money
)
as
begin
select @para1-@para2
from yedek_parca

end





go 

create procedure yeni_kayit
(

@musteri_adi varchar(30),
@musteri_soyadi varchar(30),
@adres varchar(40),
@tel int
)
as
if EXISTS(SELECT * FROM musteri WHERE  @tel=tel)
print 'bu numara kayıtlıdır'
begin 
insert into musteri values
(

@musteri_adi,
@musteri_soyadi,
@adres,
@tel 

)
end



go 
create procedure kayit_silme
(
@musteri_no int
)
as
begin
delete musteri
where @musteri_no=musteri_no
end




go
create procedure toplam_kar
as
begin
select sum(alinan_para)-sum(verilen_para) 
from para 
end


go 
create procedure bilgi_goster
as 
begin
SELECT *
FROM musteri
INNER JOIN arac
ON musteri.musteri_no = arac.musteri_no;

end

go 
create procedure ortalama
as
begin
SELECT AVG(bakim_ucreti)
FROM bakim
end


go 
create procedure yapilan_son_bakim
as
begin 
SELECT MAX(bakim_bitis_tarih)
FROM bakim
end


go 
create procedure para_sirala
as
begin
SELECT alinan_para
FROM para

GROUP BY musteri_no,alinan_para
ORDER BY musteri_no desc

end



go 
create procedure bilgi_güncelleme
(
@musteri_no int,
@tel varchar(11) 
)
as
begin
UPDATE musteri
SET tel = @tel
WHERE @musteri_no =musteri_no;
end

on yedek_parca
after 
insert
as
begin

print 'yapılan satışta kar yoktur'
end

CREATE trigger yeniTutar
on yedek_parca
after
insert 
as
begin
UPDATE yedek_parca
SET parca_satis_fiyati=parca_satis_fiyati+((parca_satis_fiyati*40)/100)
print 'zam sonrası yeni fiyatlar';
END
go

on para
after 
insert
as 
begin
if exists (select 1 from para where verilen_para>alinan_para)
print 'eksik ödeme alınmıştır'
end
